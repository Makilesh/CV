# CPU-only dependency set for the containerised replay/eval path.
#
# Deliberately NOT the root requirements.txt: no torch, no onnxruntime-gpu, no timm, no
# transformers/bitsandbytes. Those exist to produce the measured numbers on the specific machine
# described in RESULTS.md, and this container must not appear to reproduce them.
#
# TORCH IS ABSENT ON PURPOSE, and it is verified rather than assumed: every `import torch` in the
# package is inside a function, and the two on the eval path (runner.seed_everything,
# metrics._environment) are wrapped so a missing torch is recorded as `"torch": null` plus the
# reason, never as a fabricated value. Running the CI selection with torch, torchvision, timm,
# onnxruntime, transformers and bitsandbytes all blocked by an import hook passes 179/179.
#
# It also avoids a trap: on Linux the default PyPI `torch` wheel is the ~2.5 GB CUDA build, so a
# plain `torch==2.11.0` pin would pull CUDA into a container whose whole point is that it does not
# have any.
numpy==2.5.1
opencv-python-headless==5.0.0.93
hydra-core==1.3.4
omegaconf==2.3.1
matplotlib==3.11.1
